
=================== SUCCESS! With Upstream ===================

# ------------------------------------------------------------
# ohmy.zeabur.app
# ------------------------------------------------------------



map $scheme $hsts_header {
    https   "max-age=63072000; preload";
}

{# ============ 负载均衡 upstream 配置 ============ #}



{# 同构协议模式：标准 upstream 块 #}
upstream npm_upstream_1 {
  
  

  
  server mirror.zeabur.app:80 weight=1;
  
}




server {

  {# 负载均衡模式：使用 upstream 名称 #}
  set $forward_scheme http;
  set $server         "npm_upstream_1";
  set $port           80;


  listen 80;
#listen [::]:80;


  server_name ohmy.zeabur.app;
http2 off;












  access_log /data/logs/proxy-host-1_access.log proxy;
  error_log /data/logs/proxy-host-1_error.log warn;







  location / {






    


    {# 负载均衡模式：直接 proxy_pass 到 upstream #}
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-Scheme $scheme;
    proxy_set_header X-Forwarded-Proto  $scheme;
    proxy_set_header X-Forwarded-For    $remote_addr;
    proxy_set_header X-Real-IP          $remote_addr;
    proxy_pass http://npm_upstream_1;

  }


  # Custom
  include /data/nginx/custom/server_proxy[.]conf;
}
